"""
Plantilla para crear nuevos tests de endpoints de FastAPI.
Copia este archivo y adapta los tests a tus necesidades.
"""
import pytest
from fastapi.testclient import TestClient
from api.main import app

client = TestClient(app)


@pytest.mark.api
class TestNuevoEndpoint:
    """Tests para un nuevo endpoint"""

    def test_endpoint_get(self):
        """Test GET básico"""
        response = client.get("/api/tu-endpoint")
        assert response.status_code == 200
        assert "key" in response.json()

    def test_endpoint_post(self):
        """Test POST con datos"""
        data = {"campo1": "valor1", "campo2": "valor2"}
        response = client.post("/api/tu-endpoint", json=data)
        assert response.status_code == 201
        assert response.json()["campo1"] == "valor1"

    def test_endpoint_put(self):
        """Test PUT para actualizar"""
        data = {"campo1": "nuevo_valor"}
        response = client.put("/api/tu-endpoint/1", json=data)
        assert response.status_code == 200

    def test_endpoint_delete(self):
        """Test DELETE"""
        response = client.delete("/api/tu-endpoint/1")
        assert response.status_code == 204

    def test_endpoint_validation(self):
        """Test de validación de datos"""
        # Datos inválidos
        data = {"campo_invalido": "valor"}
        response = client.post("/api/tu-endpoint", json=data)
        assert response.status_code == 422  # Unprocessable Entity

    def test_endpoint_not_found(self):
        """Test de recurso no encontrado"""
        response = client.get("/api/tu-endpoint/999999")
        assert response.status_code == 404

    def test_endpoint_authentication(self):
        """Test de autenticación requerida"""
        # Sin token
        response = client.get("/api/protected-endpoint")
        assert response.status_code == 401

        # Con token válido
        headers = {"Authorization": "Bearer token_valido"}
        response = client.get("/api/protected-endpoint", headers=headers)
        assert response.status_code == 200


@pytest.mark.integration
class TestIntegracionCompleta:
    """Tests de integración que involucran múltiples operaciones"""

    def test_flujo_completo(self):
        """Test de un flujo completo CRUD"""
        # 1. Crear
        data = {"nombre": "Test"}
        response = client.post("/api/items", json=data)
        assert response.status_code == 201
        item_id = response.json()["id"]

        # 2. Leer
        response = client.get(f"/api/items/{item_id}")
        assert response.status_code == 200
        assert response.json()["nombre"] == "Test"

        # 3. Actualizar
        data = {"nombre": "Test Actualizado"}
        response = client.put(f"/api/items/{item_id}", json=data)
        assert response.status_code == 200

        # 4. Eliminar
        response = client.delete(f"/api/items/{item_id}")
        assert response.status_code == 204


# Ejemplo de uso de fixtures personalizados
@pytest.fixture
def item_creado():
    """Fixture que crea un item para usar en tests"""
    data = {"nombre": "Item de Prueba"}
    response = client.post("/api/items", json=data)
    item = response.json()
    yield item
    # Cleanup: eliminar el item después del test
    client.delete(f"/api/items/{item['id']}")


def test_con_fixture(item_creado):
    """Test que usa un fixture personalizado"""
    response = client.get(f"/api/items/{item_creado['id']}")
    assert response.status_code == 200
    assert response.json()["nombre"] == "Item de Prueba"


# Tests parametrizados
@pytest.mark.parametrize("input_data,expected_status", [
    ({"nombre": "valido"}, 201),
    ({"nombre": ""}, 422),  # Nombre vacío
    ({}, 422),  # Sin datos
    ({"campo_invalido": "valor"}, 422),
])
def test_validacion_parametrizada(input_data, expected_status):
    """Test parametrizado para diferentes casos de validación"""
    response = client.post("/api/items", json=input_data)
    assert response.status_code == expected_status
